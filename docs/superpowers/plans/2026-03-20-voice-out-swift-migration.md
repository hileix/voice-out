```swift
import Foundation

/// Represents a keyboard shortcut with modifiers and a key
struct ShortcutModel: Codable, Equatable {
    var key: String
    var modifiers: Modifiers

    struct Modifiers: Codable, Equatable {
        var control: Bool
        var shift: Bool
        var alt: Bool
        var meta: Bool

        var carbonModifiers: UInt32 {
            var carbon: UInt32 = 0
            if control { carbon |= UInt32(controlKey) }
            if shift { carbon |= UInt32(shiftKey) }
            if alt { carbon |= UInt32(optionKey) }
            if meta { carbon |= UInt32(cmdKey) }
            return carbon
        }

        var displayText: String {
            var parts: [String] = []
            if control { parts.append("⌃") }
            if alt { parts.append("⌥") }
            if shift { parts.append("⇧") }
            if meta { parts.append("⌘") }
            return parts.joined()
        }

        var hasAnyModifier: Bool {
            return control || shift || alt || meta
        }
    }

    /// Default shortcut: Ctrl+S
    static let `default` = ShortcutModel(
        key: "s",
        modifiers: Modifiers(control: true, shift: false, alt: false, meta: false)
    )

    var displayText: String {
        let keyDisplay = key == " " ? "Space" : key.uppercased()
        return modifiers.displayText + keyDisplay
    }

    var isValid: Bool {
        // Must have at least one modifier
        guard modifiers.hasAnyModifier else { return false }
        // Key must not be empty and not be a modifier key itself
        guard !key.isEmpty else { return false }
        let modifierKeys = ["control", "shift", "alt", "option", "command", "meta"]
        return !modifierKeys.contains(key.lowercased())
    }
}

// MARK: - UserDefaults Extension

extension ShortcutModel {
    private static let userDefaultsKey = "globalShortcut"

    /// Load shortcut from UserDefaults, fallback to default if not found or invalid
    static func loadFromUserDefaults() -> ShortcutModel {
        guard let data = UserDefaults.standard.data(forKey: userDefaultsKey),
              let shortcut = try? JSONDecoder().decode(ShortcutModel.self, from: data),
              shortcut.isValid else {
            return .default
        }
        return shortcut
    }

    /// Save shortcut to UserDefaults
    func saveToUserDefaults() {
        guard isValid,
              let data = try? JSONEncoder().encode(self) else {
            return
        }
        UserDefaults.standard.set(data, forKey: Self.userDefaultsKey)
    }
}

// Carbon API constants (from HIToolbox framework)
private let controlKey: Int = 1 << 12
private let shiftKey: Int = 1 << 9
private let optionKey: Int = 1 << 11
private let cmdKey: Int = 1 << 8
```

import Foundation
import AppKit

/// Manages Accessibility permissions required for reading selected text
class PermissionsManager: ObservableObject {
    @Published private(set) var hasAccessibilityPermission: Bool = false

    static let shared = PermissionsManager()

    private init() {
        checkPermissionStatus()
        // Start polling for permission changes
        startMonitoring()
    }

    /// Check current Accessibility permission status
    func checkPermissionStatus() {
        let trusted = AXIsProcessTrusted()
        DispatchQueue.main.async {
            self.hasAccessibilityPermission = trusted
        }
    }

    /// Request Accessibility permission (shows system prompt)
    func requestPermission() {
        let options = [kAXTrustedCheckOptionPrompt.takeUnretainedValue() as String: true]
        AXIsProcessTrustedWithOptions(options as CFDictionary)

        // Check again after a short delay
        DispatchQueue.main.asyncAfter(deadline: .now() + 0.5) {
            self.checkPermissionStatus()
        }
    }

    /// Open System Settings to Accessibility privacy pane
    func openSystemSettings() {
        if let url = URL(string: "x-apple.systempreferences:com.apple.preference.security?Privacy_Accessibility") {
            NSWorkspace.shared.open(url)
        }
    }

    private var monitorTimer: Timer?

    private func startMonitoring() {
        // Poll permission status every 2 seconds
        monitorTimer = Timer.scheduledTimer(withTimeInterval: 2.0, repeats: true) { [weak self] _ in
            self?.checkPermissionStatus()
        }
    }

    deinit {
        monitorTimer?.invalidate()
    }
}

import Foundation
import AppKit
import ApplicationServices

/// Service for capturing selected text from any application
class TextSelectionService {

    /// Capture currently selected text using Accessibility API
    /// - Parameter delay: Delay in milliseconds before capturing (default 200ms)
    /// - Returns: Selected text, or nil if none available
    func getSelectedText(delayMs: Int = 200) async -> String? {
        // Wait for selection to be ready
        if delayMs > 0 {
            try? await Task.sleep(nanoseconds: UInt64(delayMs) * 1_000_000)
        }

        // Try Accessibility API first
        if let text = getTextViaAccessibility() {
            return text
        }

        // Fallback to AppleScript
        if let text = getTextViaAppleScript() {
            return text
        }

        return nil
    }

    private func getTextViaAccessibility() -> String? {
        // Get system-wide accessibility element
        let systemWide = AXUIElementCreateSystemWide()

        // Get focused element
        var focusedElement: CFTypeRef?
        let focusedResult = AXUIElementCopyAttributeValue(
            systemWide,
            kAXFocusedUIElementAttribute as CFString,
            &focusedElement
        )

        guard focusedResult == .success,
              let focused = focusedElement else {
            return nil
        }

        // Get selected text from focused element
        var selectedText: CFTypeRef?
        let textResult = AXUIElementCopyAttributeValue(
            focused as! AXUIElement,
            kAXSelectedTextAttribute as CFString,
            &selectedText
        )

        guard textResult == .success,
              let text = selectedText as? String,
              !text.trimmingCharacters(in: .whitespacesAndNewlines).isEmpty else {
            return nil
        }

        return text
    }

    private func getTextViaAppleScript() -> String? {
        let script = """
        tell application "System Events"
            try
                set selectedText to value of attribute "AXSelectedText" of focused element of (first application process whose frontmost is true)
                return selectedText
            on error
                return ""
            end try
        end tell
        """

        var error: NSDictionary?
        guard let scriptObject = NSAppleScript(source: script) else {
            return nil
        }

        let output = scriptObject.executeAndReturnError(&error)

        guard error == nil,
              let text = output.stringValue,
              !text.trimmingCharacters(in: .whitespacesAndNewlines).isEmpty else {
            return nil
        }

        return text
    }
}

import Foundation
import AppKit

/// Service for text-to-speech using NSSpeechSynthesizer
class TTSService: NSObject, NSSpeechSynthesizerDelegate {
    private var synthesizer: NSSpeechSynthesizer?
    private var currentlySpeaking: Bool = false

    override init() {
        super.init()
        initializeSynthesizer()
    }

    private func initializeSynthesizer() {
        synthesizer = NSSpeechSynthesizer()
        synthesizer?.delegate = self
    }

    /// Speak the given text
    /// - Parameter text: Text to speak
    func speak(_ text: String) {
        guard !text.trimmingCharacters(in: .whitespacesAndNewlines).isEmpty else {
            print("[TTSService] Empty text, skipping speech")
            return
        }

        guard let synthesizer = synthesizer else {
            print("[TTSService] Error: Synthesizer not initialized")
            return
        }

        // Stop current speech if any
        if currentlySpeaking {
            synthesizer.stopSpeaking()
        }

        // Start speaking
        print("[TTSService] Speaking: \(text.prefix(50))...")
        currentlySpeaking = true
        synthesizer.startSpeaking(text)
    }

    /// Stop current speech
    func stop() {
        synthesizer?.stopSpeaking()
        currentlySpeaking = false
    }

    // MARK: - NSSpeechSynthesizerDelegate

    func speechSynthesizer(_ sender: NSSpeechSynthesizer, didFinishSpeaking finishedSpeaking: Bool) {
        currentlySpeaking = false
        print("[TTSService] Finished speaking")
    }
}

```swift
import Foundation
import Carbon
import AppKit

/// Manages global keyboard shortcuts using Carbon API
class ShortcutManager {
    private var hotKeyRef: EventHotKeyRef?
    private var eventHandler: EventHandlerRef?
    private var onShortcutTriggered: (() -> Void)?

    /// Register a global keyboard shortcut
    /// - Parameters:
    ///   - shortcut: The shortcut to register
    ///   - callback: Closure to call when shortcut is triggered
    /// - Returns: true if registration succeeded, false otherwise
    func registerShortcut(_ shortcut: ShortcutModel, onTriggered callback: @escaping () -> Void) -> Bool {
        // Unregister existing shortcut first
        unregisterShortcut()

        guard shortcut.isValid else {
            print("[ShortcutManager] Invalid shortcut")
            return false
        }

        onShortcutTriggered = callback

        // Convert key to Carbon keycode
        guard let keyCode = carbonKeyCodeForKey(shortcut.key) else {
            print("[ShortcutManager] Could not convert key '\(shortcut.key)' to keycode")
            return false
        }

        // Register Carbon event handler
        var eventType = EventTypeSpec(eventClass: OSType(kEventClassKeyboard), eventKind: UInt32(kEventHotKeyPressed))
        let status = InstallEventHandler(
            GetApplicationEventTarget(),
            { (_, event, userData) -> OSStatus in
                guard let userData = userData else { return OSStatus(eventNotHandledErr) }
                let manager = Unmanaged<ShortcutManager>.fromOpaque(userData).takeUnretainedValue()
                manager.handleHotKeyEvent()
                return noErr
            },
            1,
            &eventType,
            Unmanaged.passUnretained(self).toOpaque(),
            &eventHandler
        )

        guard status == noErr else {
            print("[ShortcutManager] Failed to install event handler: \(status)")
            return false
        }

        // Register hot key
        let hotKeyID = EventHotKeyID(signature: OSType(0x56434F55), id: 1) // 'VCOU' = VoiceOut
        let registerStatus = RegisterEventHotKey(
            UInt32(keyCode),
            shortcut.modifiers.carbonModifiers,
            hotKeyID,
            GetApplicationEventTarget(),
            0,
            &hotKeyRef
        )

        guard registerStatus == noErr else {
            print("[ShortcutManager] Failed to register hotkey: \(registerStatus)")
            RemoveEventHandler(eventHandler)
            eventHandler = nil
            return false
        }

        print("[ShortcutManager] Registered shortcut: \(shortcut.displayText)")
        return true
    }

    /// Unregister current shortcut
    func unregisterShortcut() {
        if let hotKeyRef = hotKeyRef {
            UnregisterEventHotKey(hotKeyRef)
            self.hotKeyRef = nil
        }

        if let eventHandler = eventHandler {
            RemoveEventHandler(eventHandler)
            self.eventHandler = nil
        }

        onShortcutTriggered = nil
    }

    private func handleHotKeyEvent() {
        onShortcutTriggered?()
    }

    private func carbonKeyCodeForKey(_ key: String) -> Int? {
        // Map keys to Carbon keycodes (comprehensive mapping)
        let keyMap: [String: Int] = [
            // Letters
            "a": 0, "s": 1, "d": 2, "f": 3, "h": 4, "g": 5, "z": 6, "x": 7, "c": 8, "v": 9,
            "b": 11, "q": 12, "w": 13, "e": 14, "r": 15, "y": 16, "t": 17,
            "o": 31, "u": 32, "i": 34, "p": 35, "l": 37, "j": 38, "k": 40, "n": 45, "m": 46,
            // Numbers
            "1": 18, "2": 19, "3": 20, "4": 21, "5": 23, "6": 22, "7": 26, "8": 28, "9": 25, "0": 29,
            // Punctuation
            "-": 27, "=": 24, "[": 33, "]": 30, "\\": 42, ";": 41, "'": 39, ",": 43, ".": 47, "/": 44, "`": 50,
            // Special keys
            " ": 49, // space
            "return": 36, "tab": 48, "delete": 51, "escape": 53,
            // Arrow keys
            "left": 123, "right": 124, "down": 125, "up": 126,
            // Function keys
            "f1": 122, "f2": 120, "f3": 99, "f4": 118, "f5": 96, "f6": 97,
            "f7": 98, "f8": 100, "f9": 101, "f10": 109, "f11": 103, "f12": 111,
        ]

        let normalized = key.lowercased()
        if let code = keyMap[normalized] {
            return code
        }

        // If key not found, log warning
        print("[ShortcutManager] Warning: No keycode mapping for key '\(key)'")
        return nil
    }

    deinit {
        unregisterShortcut()
    }
}

import Foundation
import Combine

/// Main app view model coordinating all services
class AppViewModel: ObservableObject {
    @Published var currentShortcut: ShortcutModel
    @Published var isRecordingShortcut = false

    private let shortcutManager = ShortcutManager()
    private let textSelectionService = TextSelectionService()
    private let ttsService = TTSService()
    let permissionsManager = PermissionsManager.shared

    init() {
        // Load saved shortcut
        currentShortcut = ShortcutModel.loadFromUserDefaults()

        // Register shortcut
        registerCurrentShortcut()
    }

    /// Register the current shortcut
    private func registerCurrentShortcut() {
        let success = shortcutManager.registerShortcut(currentShortcut) { [weak self] in
            self?.handleShortcutTriggered()
        }

        if !success {
            print("[AppViewModel] Failed to register shortcut")
        }
    }

    /// Handle shortcut triggered event
    private func handleShortcutTriggered() {
        Task {
            await readAndSpeakSelectedText()
        }
    }

    /// Read selected text and speak it
    private func readAndSpeakSelectedText() async {
        guard permissionsManager.hasAccessibilityPermission else {
            print("[AppViewModel] Accessibility permission not granted")
            return
        }

        guard let selectedText = await textSelectionService.getSelectedText() else {
            print("[AppViewModel] No text selected")
            return
        }

        ttsService.speak(selectedText)
    }

    /// Update shortcut (called from settings)
    func updateShortcut(_ newShortcut: ShortcutModel) {
        guard newShortcut.isValid else {
            print("[AppViewModel] Invalid shortcut, not updating")
            return
        }

        currentShortcut = newShortcut
        currentShortcut.saveToUserDefaults()
        registerCurrentShortcut()
    }

    /// Reset to default shortcut
    func resetToDefaultShortcut() {
        updateShortcut(.default)
    }
}
```

import SwiftUI
import AppKit

/// Interactive shortcut recorder control using NSEvent monitoring
struct ShortcutRecorderView: View {
    @Binding var shortcut: ShortcutModel
    @Binding var isRecording: Bool
    @State private var monitor: Any?

    var body: some View {
        VStack(alignment: .leading, spacing: 8) {
            Text("Keyboard Shortcut")
                .font(.headline)

            ZStack {
                RoundedRectangle(cornerRadius: 8)
                    .fill(isRecording ? Color.blue.opacity(0.1) : Color.gray.opacity(0.1))
                    .overlay(
                        RoundedRectangle(cornerRadius: 8)
                            .stroke(isRecording ? Color.blue : Color.gray, lineWidth: 2)
                    )

                Text(isRecording ? "Press shortcut keys..." : shortcut.displayText)
                    .font(.system(.body, design: .monospaced))
                    .foregroundColor(isRecording ? .blue : .primary)
                    .padding()
            }
            .frame(height: 60)
            .onTapGesture {
                startRecording()
            }

            Text("Click to record, then press your desired shortcut combination")
                .font(.caption)
                .foregroundColor(.secondary)
        }
        .onDisappear {
            stopRecording()
        }
    }

    private func startRecording() {
        isRecording = true

        // Use NSEvent local monitor to capture key presses
        monitor = NSEvent.addLocalMonitorForEvents(matching: .keyDown) { event in
            handleKeyEvent(event)
            return nil // Consume the event
        }
    }

    private func stopRecording() {
        if let monitor = monitor {
            NSEvent.removeMonitor(monitor)
            self.monitor = nil
        }
        isRecording = false
    }

    private func handleKeyEvent(_ event: NSEvent) {
        guard isRecording else { return }

        // Get the key character
        guard let characters = event.charactersIgnoringModifiers,
              let firstChar = characters.first else {
            return
        }

        let key = String(firstChar).lowercased()

        // Get modifiers
        let modifierFlags = event.modifierFlags
        let modifiers = ShortcutModel.Modifiers(
            control: modifierFlags.contains(.control),
            shift: modifierFlags.contains(.shift),
            alt: modifierFlags.contains(.option),
            meta: modifierFlags.contains(.command)
        )

        // Build new shortcut
        let newShortcut = ShortcutModel(key: key, modifiers: modifiers)

        // Validate and update if valid
        if newShortcut.isValid {
            shortcut = newShortcut
            stopRecording()
        }
    }
}
